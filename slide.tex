\section*{Complexity}
\begin{frame}

	\begin{itemize}
		\item The mathematical methods of complexity begun to be developed in the '70s of the XIX century

	\centering
	A system is complex if and only if it must be described differently on different levels of one or more scales

\begin{frame}
	\frametitle{The Base Model}
	$$p = (1 + \mu) \frac{W}{Y} = (1+\mu)\beta$$
	$$C^T = \frac{a_Y ((1 - \tau_W) W + UB) + a_V {\mathbf{M}_\mathcal{H}}_{t-1}}{(1 + \tau_C) p_{t-1}}$$
	$$G^T = \frac{d \cdot {GDP}_{t-1} + T_{t-1} - UB_{t-1}}{p_{t-1}}$$
	$$N = Y^T/\beta$$
	$$Y = \beta N$$
\end{frame}

\begin{frame}
	\frametitle{Circuit}
	$$\Delta^+ \mathbf{L}_{\mathcal{F}_\mathbf{C}} = W_{\mathcal{F}_\mathbf{C}} + p_\mathbf{K} I_{\mathcal{F}_\mathbf{C}}$$ $$\Delta^+\mathbf{L}_{\mathcal{F}_\mathbf{K}} = W_{\mathcal{F}_\mathbf{K}}$$
	$$\mathbf{L}_{\mathcal{F}_\mathbf{C}} = {\mathbf{L}_{\mathcal{F}_\mathbf{C}}}_{t-1} + \Delta^+\mathbf{L}_{\mathcal{F}_\mathbf{C}} - {\mathbf{M}_{\mathcal{F}_\mathbf{C}}} - p_\mathbf{C} (C + G)$$ $$\mathbf{L}_{\mathcal{F}_\mathbf{K}} = {\mathbf{L}_{\mathcal{F}_\mathbf{K}}}_{t-1} + \Delta^+\mathbf{L}_{\mathcal{F}_\mathbf{K}} - {\mathbf{M}_{\mathcal{F}_\mathbf{K}}} - p_\mathbf{K} I_{\mathcal{F}_\mathbf{C}}$$
	$$\mathbf{B} = \frac{\mathbf{B}_{t-1} - (T - p_\mathbf{C}G - UB)}{1-r_\mathbf{B}}$$
	$$r_\mathbf{B} = i_{t-1} + \alpha_i (i_{i-1} - i^T) + \alpha_{cu} ({cu}_{t-1} - {cu}^T) - \alpha_u (u_{t-1} - u^T)$$ 
\end{frame}

\begin{frame}
	\frametitle{Innovation}
	\centering
	$$W_0 =  {W_0}_{t-1} (1 + \Theta_W \frac{N^T_{t-1} - N_{t-1}}{N^T_{t-1}})$$
	$$\beta = \beta_{t-1} + \iota_1 <\sigma> + \iota_2 \frac{p_\mathbf{K}I_{\mathcal{F}_\mathbf{C}}}{\Sigma p_\mathbf{K}I_{\mathcal{F}_\mathbf{C}}} - \iota_3 age$$
	$\sigma_{t-1} = \sigma (1+\frac{\delta_\sigma}{D})$
	if the worker is employed \\
	$\sigma_{t-1} = \max(1,\sigma / (1+\delta_\sigma))$ if the worker is unemployed
	$$W = W_0 \sigma$$
\end{frame}

\section*{Conclusion}
\begin{frame}
	\frametitle{Conclusion}
	\begin{itemize}
		\item Complexity is still an ill-defined word
		\item A phenomenological approach would shake the foundations of the discipline
		\item A step-by-step approach makes the development feasible
		\item Even in such a simple model, an UBI has no clear policy implication and heavily depends on the context
	\end{itemize}
\end{frame}
