\chapter{The Bank's Profits equation}
\label{app:eq}
Bank aims to reach a target credit rate
$${cr}^T = \frac{V_\mathcal{B}}{\mathbf{L}} = \frac{\mathbf{L} + \mathbf{B} - \mathbf{M}}{\mathbf{L}}$$
$$\mathbf{M} = (1-{cr}^T) \mathbf{L} + \mathbf{B}$$
But from the model accounting we also have (as the sum of $\mathbf{M}_\mathcal{H}$, $\mathbf{M}_{\mathcal{F}_\mathbf{C}}$, $\mathbf{M}_{\mathcal{F}_\mathbf{K}}$ eliminating the terms with opposite signs)
$$\mathbf{M} = \mathbf{M}_{t-1} + P_\mathcal{B} + UB + p_\mathbf{C} G - T -r_\mathbf{L}(\Delta^+\mathbf{L} + \mathbf{L}_{t-1}) + (\mathbf{L} - \mathbf{L}_{t-1})$$
$$\mathbf{B} = \frac{\mathbf{B}_{t-1} - (T - p_\mathbf{C}G - UB)}{1-r_\mathbf{B}}$$
Putting together and simplifying
$$\mathbf{M}_{t-1} + P_\mathcal{B} + UB + p_\mathbf{C} G - T -r_\mathbf{L}(\Delta^+\mathbf{L} + \mathbf{L}_{t-1}) + (\mathbf{L} - \mathbf{L}_{t-1}) = (1-{cr}^T) \mathbf{L} + \frac{\mathbf{B}_{t-1} - (T - p_\mathbf{C}G - UB)}{1-r_\mathbf{B}}$$
$$\mathbf{M}_{t-1} + P_\mathcal{B} - r_\mathbf{L} (\Delta^+\mathbf{L} + \mathbf{L}_{t-1}) - \mathbf{L}_{t-1} - \mathbf{B}_{t-1} = - {cr}^T \mathbf{L} + \frac{r_\mathbf{B}}{1-r_\mathbf{B}}(\mathbf{B}_{t-1} + p_\mathbf{C} G + UB - \tau_C C - \tau_W W - \tau_P (P_{\mathcal{F}_\mathbf{C}} + P_{\mathcal{F}_\mathbf{K}} + P_\mathcal{B}))$$
$$ (1+\frac{r_\mathbf{B} \tau_P}{1-r_\mathbf{B}}) P_\mathcal{B}  = \mathbf{V}_{t-1} - {cr}^T \mathbf{L} + r_\mathbf{L} (\Delta^+\mathbf{L} + \mathbf{L}_{t-1}) + \frac{r_\mathbf{B}}{1-r_\mathbf{B}}(\mathbf{B}_{t-1} + p_\mathbf{C} G + UB - \tau_C C - \tau_W W - \tau_P (P_{\mathcal{F}_\mathbf{C}} + P_{\mathcal{F}_\mathbf{K}}))$$
and finally
$$ P_\mathcal{B}  = \frac{1-r_\mathbf{B}}{1-r_\mathbf{B}+r_\mathbf{B}\tau_P}(\mathbf{V}_{t-1} - {cr}^T \mathbf{L} + r_\mathbf{L} (\Delta^+\mathbf{L} + \mathbf{L}_{t-1}) + \frac{r_\mathbf{B}}{1-r_\mathbf{B}}(\mathbf{B}_{t-1} + p_\mathbf{C} G + UB - \tau_C C - \tau_W W - \tau_P (P_{\mathcal{F}_\mathbf{C}} + P_{\mathcal{F}_\mathbf{K}})))$$
